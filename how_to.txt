1.) python3 -m venv .venv
2.) source .venv/bin/activate
3.) python -m pip install --upgrade pip
4.) python -m pip install -r requirements.txt
5.) deactivate
6.) ./.venv/bin/python -m wardogs_gateway --config gateway.toml db-init
7.) key management:

# create admin key:
./.venv/bin/python -m wardogs_gateway --config gateway.toml key-create --name "<USER-NAME>" --role admin

# create FULL admin key:
./.venv/bin/python -m wardogs_gateway --config gateway.toml key-create --name "<USER-NAME>" --role full_admin

# show keys:
./.venv/bin/python -m wardogs_gateway --config gateway.toml key-list

# deactivate key:
./.venv/bin/python -m wardogs_gateway --config gateway.toml key-revoke <KEY>

# reactivate key:
./.venv/bin/python -m wardogs_gateway --config gateway.toml key-enable <KEY>

# regenerate key for a user:
./.venv/bin/python -m wardogs_gateway --config gateway.toml key-rotate <KEY>

# change key role:
./.venv/bin/python -m wardogs_gateway --config gateway.toml key-set-role <KEY> <ROLLE> (full_admin oder admin)

# delete key:
./.venv/bin/python -m wardogs_gateway --config gateway.toml key-delete <KEY>
